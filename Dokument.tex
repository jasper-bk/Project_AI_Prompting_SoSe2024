    This project explores the use of generative AI, specifically GPT-4 and DALL-E, to create hyper-personalized product pages for websites, emails, and other digital marketing channels.

    Future goals involve expanding personalization capabilities, enhancing AI models' context awareness, and exploring integration with emerging technologies. 
    Continuous improvements will focus on advanced data analytics, model enhancement, and maintaining ethical standards.

        import os
        import dotenv
        import langchain
        
        from langchain_core.prompts import ChatPromptTemplate
        from langchain_core.output_parsers import StrOutputParser
        from langchain_core.runnables import RunnablePassthrough
        from langchain_openai import AzureChatOpenAI, AzureOpenAIEmbeddings

        dotenv.load_dotenv()

        # Langchain flags 
        langchain.verbose = False
        langchain.debug = False
        langchain.llm_cache = False

        class LangChainChat:
            def __init__(self) -> None:
                # Set up Azure OpenAI client
                self.llm = AzureChatOpenAI(
                openai_api_key=os.getenv('LLM_API_KEY'),
                azure_endpoint=os.getenv('LLM_ENDPOINT'),
                openai_api_version=os.getenv('LLM_VERSION'),
                deployment_name=os.getenv('LLM_DEPLOYMENT_NAME')
                )

                # Parameters for the language model
                self.llm.temperature=0.5
                self.llm.max_tokens=100

            def chat_bot(self, template: str, question : str) -> str:

                prompt = ChatPromptTemplate.from_template(template)
            
                chain = (
                    {
                        "question": RunnablePassthrough()
                    }
                    | prompt
                    | self.llm
                    | StrOutputParser()
                )
            
                print("Send request to the language model...")
            
                response = chain.invoke(question)
            
                print(f"Question: {question}")
                print(f"Response: {response}")

                return response

        def main() -> None:
            template : str = """
            You are an highly personalized marketing assistant and answers questions to our product.

            Question: {question}
            """
        
            question : str = "Generate a detailed product description about ..."
        
            chat = LangChainChat()
            chat.chat_bot(template, question)
